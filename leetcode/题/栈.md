使用栈即可解决，需要考虑的特殊情况：
1. 数字未必只有1位，因此在遇到数字时需要直接全部找完再压入栈；
2. char和string格式转换的问题，需要将char转换为string才能够压入栈；
3. 字符串和栈的操作函数需要掌握。

```C++
class Solution {
public:
    string decodeString(string s) {
        stack<string> code_stack;
        string res = "";
        int i = 0;
        
        while(i < s.size()){
            if(s[i] != ']'){
                if(s[i] >= '0' && s[i] <= '9'){
                    string num = "";
                    num = num + s[i];
                    i++;
                    while(s[i] >= '0' && s[i] <= '9'){
                        num = num + s[i];
                        i++;
                    }
                    code_stack.push(num);
                }
                else{
                    string str_c;
                    str_c = s[i];
                    code_stack.push(str_c);
                    ++i;
                }
            }
            else{
                string mid;
                string let = code_stack.top();
                code_stack.pop();
                while(let != "["){
                    mid = let + mid;
                    let = code_stack.top();
                    code_stack.pop();
                }
                int n = stoi(code_stack.top());
                code_stack.pop();
                let = "";
                for(int j=0;j<n;j++){
                    let = mid + let;
                }
                code_stack.push(let);
                i++;
            }
        }

        while(!code_stack.empty()){
            string str = code_stack.top();
            code_stack.pop();
            res = str + res;
        }
        return res;
    }
};
```
