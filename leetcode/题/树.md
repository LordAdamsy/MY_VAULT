
## 0.1 **根据中序和前序遍历恢复唯一的树（元素不重复）**

思路：
递归，先确定根节点，在区分左右子树分别对应的中序/前序遍历后递归求解，但写法比较繁琐

my code：

```C++
class Solution {
public:
    TreeNode* buildTree(const vector<int>& preorder, const vector<int>& inorder) {
        if(preorder.size() == 1){
            return new TreeNode(preorder[0]);
        }
        else{
            int idx_in = 0;
            for(int i=0;i<inorder.size();i++){
                if(inorder[i] == preorder[0]){
                    idx_in = i;
                    break;
                }
            }
            if(idx_in == 0){
                vector<int> arr1(preorder.begin()+1, preorder.end());
                vector<int> arr2(inorder.begin()+1, inorder.end());
                TreeNode *right = buildTree(arr1, arr2);          
                return new TreeNode(preorder[0], nullptr, right);
            }
            else if(idx_in == inorder.size()-1){
                vector<int> arr1(preorder.begin()+1, preorder.end());
                vector<int> arr2(inorder.begin(), inorder.end()-1);
                TreeNode* left =  buildTree(arr1, arr2);
                return new TreeNode(preorder[0], left, nullptr);
            }
            else{  
                vector<int> arr1(preorder.begin()+1, preorder.begin()+1+idx_in);
                vector<int> arr2(inorder.begin(), inorder.begin()+idx_in);
                TreeNode* left = buildTree(arr1, arr2);
                vector<int> arr3(preorder.begin()+1+idx_in, preorder.end());
                vector<int> arr4(inorder.begin()+idx_in+1, inorder.end());
                TreeNode* right = buildTree(arr3, arr4);
                return new TreeNode(preorder[0], left, right);    
            }
        }
    }
};
```
